// TestManagedSDKReviewedUpstream reads native source metadata through the actual
// SDK decoder and transport; the fixture never invokes declared resources.
func TestManagedSDKReviewedUpstream(t *testing.T) {
	_, corpus := newReviewedCorpus(t)
	c, err := sdk.New(sdk.Config{Origin: corpus.Origin, AllowLoopbackHTTP: true, Token: "reader"})
	must(t, err)
	defer c.Close()
	other, err := sdk.New(sdk.Config{Origin: corpus.Origin, AllowLoopbackHTTP: true, Token: "other"})
	must(t, err)
	defer other.Close()
	if len(corpus.Cases) != 6 {
		t.Fatal("incomplete reviewed corpus")
	}
	query := decode[sdk.CatalogSearch](t, []byte(`{"origins":["local"],"nativeCapabilities":{"agentSkillIds":["not-declared-sdk-sentinel"]}}`))
	selected, err := c.SearchCatalogWithResponse(t.Context(), query)
	must(t, err)
	if selected.StatusCode() != 200 || selected.JSON200 == nil || selected.JSON200.DurableTotal != 0 || len(selected.JSON200.Items) != 0 {
		t.Fatal("native SDK selectors were lost", selected.StatusCode())
	}
	var researcher string
	for _, item := range corpus.Cases {
		if item.Name == "researcher" {
			researcher = item.ReleaseID
		}
	}
	if researcher == "" {
		t.Fatal("missing independent Agent control")
	}
	for _, streaming := range []bool{true, false} {
		native := `{"agentSkillIds":["not-declared-sdk-sentinel","research"],"agentSkillTags":["planning"],"agentDefaultInputModes":["text/plain"],"agentDefaultOutputModes":["text/plain"],"agentStreaming":` + strconv.FormatBool(streaming) + `}`
		query := decode[sdk.CatalogSearch](t, []byte(`{"origins":["local"],"nativeCapabilities":`+native+`}`))
		selected, err := c.SearchCatalogWithResponse(t.Context(), query)
		must(t, err)
		want := 0
		if streaming {
			want = 1
		}
		if selected.StatusCode() != 200 || selected.JSON200 == nil || selected.JSON200.DurableTotal != want || len(selected.JSON200.Items) != want {
			t.Fatal("native SDK selection", selected.StatusCode(), streaming)
		}
		if want == 1 && (selected.JSON200.Items[0].Local == nil || selected.JSON200.Items[0].Local.Release.Id.String() != researcher) {
			t.Fatal("native SDK selected a different release")
		}
		facets := decode[sdk.ExploreCatalogJSONRequestBody](t, []byte(`{"query":{"nativeCapabilities":`+native+`},"facets":[{"field":"kind"}]}`))
		explored, err := c.ExploreCatalogWithResponse(t.Context(), facets)
		must(t, err)
		if explored.StatusCode() != 200 || explored.JSON200 == nil || explored.JSON200.Matched != want {
			t.Fatal("native SDK Explore population", explored.StatusCode(), streaming)
		}
	}
	for _, expected := range corpus.Cases {
		t.Run(expected.Name, func(t *testing.T) {
			id := decode[sdk.CatalogRecordId](t, []byte(strconv.Quote(expected.ReleaseID)))
			reply, err := c.GetCatalogRecordWithResponse(t.Context(), id)
			must(t, err)
			if reply.StatusCode() != 200 || reply.JSON200 == nil {
				t.Fatalf("reviewed record: HTTP %d", reply.StatusCode())
			}
			encoded, err := json.Marshal(reply.JSON200)
			must(t, err)
			record := decode[map[string]any](t, encoded)
			checkReviewedRecord(t, expected, record)
			if expected.Name == "hf-cli" {
				rep := reply.JSON200.VersionRevision.Representations[0]
				if rep.ArtifactId == nil || expected.Text == "" {
					t.Fatal("missing retained Skill text")
				}
				artifact, err := c.GetArtifactContentWithResponse(t.Context(), *rep.ArtifactId)
				must(t, err)
				if artifact.StatusCode() != 200 || string(artifact.Body) != expected.Text {
					t.Fatal("Unicode Skill bytes changed")
				}
				headers := artifact.Headers200
				if headers == nil || headers.ContentDigest == nil || *headers.ContentDigest != expected.ArtifactContentDigest ||
					headers.ContentLength == nil || *headers.ContentLength != strconv.Itoa(len(expected.Text)) ||
					artifact.HTTPResponse.Header.Get("Content-Type") != expected.Representation["mediaType"] {
					t.Fatal("Skill download headers do not describe the source bytes")
				}
				hidden, err := other.GetArtifactContentWithResponse(t.Context(), *rep.ArtifactId)
				must(t, err)
				problem := hidden.ApplicationproblemJSONDefault
				if hidden.StatusCode() != 404 || problem == nil || problem.Status != 404 || problem.Code != "NOT_FOUND" ||
					problem.RequestId == "" || problem.RequestId != hidden.HTTPResponse.Header.Get("X-Request-ID") {
					t.Fatal("foreign-tenant artifact read did not return the expected correlated problem")
				}
			}
			denied, err := other.GetCatalogRecordWithResponse(t.Context(), id)
			must(t, err)
			if denied.StatusCode() != 404 {
				t.Fatal("reviewed resource crossed a tenant boundary")
			}
		})
	}
}

func TestManagedSDKRelationshipPagesAndFollow(t *testing.T) {
	_, corpus := newReviewedCorpus(t)
	cases := map[string]reviewedCase{}
	for _, item := range corpus.Cases {
		cases[item.Name] = item
	}
	source, target := cases["context7-plugin"], cases["context7-registry"]
	sourceID := decode[sdk.ReleaseId](t, []byte(strconv.Quote(source.ReleaseID)))
	c, err := sdk.New(sdk.Config{Origin: corpus.Origin, AllowLoopbackHTTP: true, Token: "follower"})
	must(t, err)
	defer c.Close()

	gaps, err := c.ListListingMetadataGapsWithResponse(t.Context(), &sdk.ListListingMetadataGapsParams{Checks: ptr("license,support"), Kind: ptr(sdk.ResourceKindPlugin)})
	must(t, err)
	if gaps.JSON200 == nil || len(gaps.JSON200.Items) != 1 || gaps.JSON200.Items[0].ReleaseId != sourceID ||
		!reflect.DeepEqual(gaps.JSON200.Items[0].Missing, []sdk.ListingMetadataGapMissing{sdk.ListingMetadataGapMissingSupport, sdk.ListingMetadataGapMissingLicense}) {
		t.Fatalf("selected Plugin Listing omissions (HTTP %d)", gaps.StatusCode())
	}

	query := &sdk.ListReleaseRelationshipsParams{View: "outgoing", Limit: ptr(1)}
	first, err := c.ListReleaseRelationshipsWithResponse(t.Context(), sourceID, query)
	must(t, err)
	if first.JSON200 == nil || len(first.JSON200.Items) != 1 || first.JSON200.NextCursor == nil || first.JSON200.Items[0].Declaration.TargetIdentifier != "fixture:unreleased-dependency" || first.JSON200.Items[0].Related != nil {
		t.Fatalf("unreleased destination or cursor changed (HTTP %d)", first.StatusCode())
	}
	query.Cursor = first.JSON200.NextCursor
	second, err := c.ListReleaseRelationshipsWithResponse(t.Context(), sourceID, query)
	must(t, err)
	if second.JSON200 == nil || len(second.JSON200.Items) != 1 || second.JSON200.NextCursor != nil {
		t.Fatalf("invalid final page (HTTP %d)", second.StatusCode())
	}
	edge := second.JSON200.Items[0]
	if edge.Declaration.TargetIdentifier != target.Identifier || edge.Related == nil || edge.Related.ReleaseId.String() != target.ReleaseID || edge.Related.Version != "4.0.3" {
		t.Fatal("exact relationship did not resolve to its current release")
	}
	listingID := decode[sdk.UUID](t, []byte(strconv.Quote(target.ListingID)))
	observed := sdk.ReleaseObservation{ReleaseId: edge.Related.ReleaseId, Revision: 1}
	body := sdk.PersonalInput{Type: "follow", Label: "Go SDK follow", ListingId: &listingID, Observed: &observed}
	saved, err := c.CreatePersonalItemWithResponse(t.Context(), &sdk.CreatePersonalItemParams{IdempotencyKey: "go-target-follow"}, body)
	must(t, err)
	if saved.JSON201 == nil || saved.JSON201.Revision != 1 {
		t.Fatalf("create follow (HTTP %d)", saved.StatusCode())
	}
	resolved, err := c.ResolvePersonalItemWithResponse(t.Context(), saved.JSON201.Id, sdk.ResolvePersonalItemJSONRequestBody{})
	must(t, err)
	if resolved.JSON200 == nil || resolved.JSON200.Change == nil || *resolved.JSON200.Change != "unchanged" || resolved.JSON200.Current == nil || *resolved.JSON200.Current != observed {
		t.Fatalf("follow did not preserve the acknowledged release (HTTP %d)", resolved.StatusCode())
	}
}

func TestSDKCompleteOperationInventory(t *testing.T) {
	var contract struct {
		Paths map[string]map[string]any `yaml:"paths"`
	}
	must(t, yaml.Unmarshal(api.ControlV1, &contract))
	generated := reflect.TypeFor[sdk.ClientInterface]()
	typed := reflect.TypeFor[sdk.ClientWithResponsesInterface]()
	seen := map[string]bool{}
	for _, path := range contract.Paths {
		for _, value := range path {
			operation, ok := value.(map[string]any)
			if !ok {
				continue
			}
			name, _ := operation["operationId"].(string)
			if name == "" {
				continue
			}
			if seen[name] {
				t.Fatal("duplicate operation ID", name)
			}
			seen[name] = true
			exported := strings.ToUpper(name[:1]) + name[1:]
			_, raw := generated.MethodByName(exported)
			if !raw {
				_, raw = generated.MethodByName(exported + "WithBody")
			}
			_, parsed := typed.MethodByName(exported + "WithResponse")
			if !parsed {
				_, parsed = typed.MethodByName(exported + "WithBodyWithResponse")
			}
			if !raw || !parsed {
				t.Fatal("omitted operation", name)
			}
		}
	}
	if len(seen) == 0 {
		t.Fatal("empty operation inventory")
	}
	t.Logf("Qualified every operation in the current unchanged contract: %d", len(seen))
}

func TestManagedSDKVersionAdoption(t *testing.T) {
	input := newVersionAdoptionCase(t)
	c, err := sdk.New(sdk.Config{Origin: input.Origin, AllowLoopbackHTTP: true, Token: "admin"})
	must(t, err)
	defer c.Close()
	observation := decode[sdk.ObservationId](t, []byte(strconv.Quote(input.Body.ObservationID)))
	resource := decode[sdk.ResourceId](t, []byte(strconv.Quote(input.ResourceID)))
	page, err := c.ListUpstreamChangesWithResponse(t.Context(), &sdk.ListUpstreamChangesParams{ObservationId: &observation})
	must(t, err)
	if page.JSON200 == nil || len(page.JSON200.Items) != 1 {
		t.Fatal("upstream inventory", page.StatusCode())
	}
	row := page.JSON200.Items[0]
	if row.Occurrence.NormalizedEntryDigest != input.NormalizedEntryDigest {
		t.Fatal("wrong normalized digest for selected occurrence", row.Occurrence)
	}
	if row.Outcome != "unadopted_version" || row.Target.ResourceId.String() != input.ResourceID || row.Occurrence.SnapshotId.String() != input.Body.SnapshotID || row.Target.PriorVersionRevisionId.String() != input.PriorVersionRevisionID {
		t.Fatal("incorrect SDK inventory handoff", row)
	}
	target, err := c.GetVersionAdoptionTargetWithResponse(t.Context(), observation)
	must(t, err)
	if target.JSON200 == nil || target.JSON200.ResourceId.String() != input.ResourceID || target.JSON200.PriorVersionRevisionId.String() != input.PriorVersionRevisionID || target.JSON200.Version != "opaque-first" || target.JSON200.UpstreamVersion != "opaque-next" {
		t.Fatal("typed adoption target changed", target.StatusCode())
	}
	encoded, err := json.Marshal(input.Body)
	must(t, err)
	body := decode[sdk.AdoptResourceVersionJSONRequestBody](t, encoded)
	params := &sdk.AdoptResourceVersionParams{IdempotencyKey: "go-enabled-adoption"}
	created, err := c.AdoptResourceVersionWithResponse(t.Context(), resource, params, body)
	must(t, err)
	if created.JSON201 == nil || created.JSON201.Version.Version != "opaque-next" || created.JSON201.VersionRevision.Adoption == nil || created.JSON201.VersionRevision.Adoption.ObservationId.String() != input.Body.ObservationID || created.JSON201.VersionRevision.Adoption.SourceId.String() != input.SourceID || created.JSON201.VersionRevision.Adoption.SnapshotId.String() != input.Body.SnapshotID || created.JSON201.VersionRevision.Adoption.RetrievalDigest != input.Body.RetrievalDigest || created.JSON201.VersionRevision.Adoption.EntryLocation != input.Body.EntryLocation || created.JSON201.ResourceId.String() != input.ResourceID || created.JSON201.Evidence.State != "complete" || len(created.JSON201.VersionRevision.Representations) != 1 || created.JSON201.VersionRevision.Representations[0].ValidationState != "valid" {
		t.Fatal("typed adoption receipt changed", created.StatusCode())
	}
	replay, err := c.AdoptResourceVersionWithResponse(t.Context(), resource, params, body)
	must(t, err)
	if replay.StatusCode() != 201 || replay.HTTPResponse.Header.Get("Idempotency-Replayed") != "true" || !bytes.Equal(created.Body, replay.Body) {
		t.Fatal("typed adoption replay changed")
	}
	params.IdempotencyKey = "go-duplicate-adoption"
	duplicate, err := c.AdoptResourceVersionWithResponse(t.Context(), resource, params, body)
	must(t, err)
	if duplicate.StatusCode() != 409 || duplicate.ApplicationproblemJSONDefault == nil || duplicate.ApplicationproblemJSONDefault.Code != "CONFLICT" {
		t.Fatal("duplicate adoption was not refused")
	}
}

func TestManagedSDKCommittedMutationRecovery(t *testing.T) {
	f := newFixture(t)
	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cancel()
	const commandKey = "sdk-lost-update"
	type committedResponse struct {
		status int
		body   []byte
		etag   string
	}
	committed := make(chan committedResponse, 1)
	var attempts atomic.Int32
	server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.Method != http.MethodPatch || r.Header.Get("Idempotency-Key") != commandKey {
			f.server.Config.Handler.ServeHTTP(w, r)
			return
		}
		if attempts.Add(1) != 1 {
			f.server.Config.Handler.ServeHTTP(w, r)
			return
		}
		// Let the real Catalog commit and retain the result. Lose only its first
		// HTTP response, so the caller cannot tell whether the command committed.
		recorded := httptest.NewRecorder()
		f.server.Config.Handler.ServeHTTP(recorded, r)
		committed <- committedResponse{recorded.Code, bytes.Clone(recorded.Body.Bytes()), recorded.Header().Get("ETag")}
		conn, _, err := w.(http.Hijacker).Hijack()
		if err != nil {
			t.Error("could not inject lost response", err)
			return
		}
		_ = conn.Close()
	}))
	t.Cleanup(server.Close)
	token := "admin"
	c, err := sdk.New(sdk.Config{Origin: server.URL, AllowLoopbackHTTP: true, TokenProvider: func(context.Context) (string, error) { return token, nil }})
	must(t, err)
	t.Cleanup(c.Close)
	ns, err := c.CreateNamespaceWithResponse(ctx, &sdk.CreateNamespaceParams{IdempotencyKey: "sdk-lost-namespace"}, sdk.CreateNamespaceJSONRequestBody{DisplayName: "Before", Slug: "sdk-lost"})
	must(t, err)
	if ns.JSON201 == nil || ns.HTTPResponse.Header.Get("ETag") == "" {
		t.Fatal("valid create control failed", ns.StatusCode())
	}
	name := "Committed once"
	params := &sdk.UpdateNamespaceParams{IdempotencyKey: commandKey, IfMatch: ns.HTTPResponse.Header.Get("ETag")}
	body := sdk.UpdateNamespaceJSONRequestBody{DisplayName: &name, Reason: "Qualify lost command response"}
	uncertain, err := c.UpdateNamespaceWithResponse(ctx, ns.JSON201.Id, params, body)
	var diagnostic *sdk.Error
	if uncertain != nil || !errors.As(err, &diagnostic) || diagnostic.Code != "request_or_decode_failed" || attempts.Load() != 1 {
		t.Fatal("lost command was retried or uncertainty was hidden", err, attempts.Load())
	}
	var original committedResponse
	select {
	case original = <-committed:
	case <-ctx.Done():
		t.Fatal("command did not reach the real Catalog", ctx.Err())
	}
	if original.status != http.StatusOK || original.etag == "" || original.etag == params.IfMatch {
		t.Fatal("fault did not follow a successful committed update", original.status)
	}
	readback, err := c.GetNamespaceWithResponse(ctx, ns.JSON201.Id)
	must(t, err)
	if readback.JSON200 == nil || readback.JSON200.DisplayName != name || readback.JSON200.Revision != ns.JSON201.Revision+1 || readback.HTTPResponse.Header.Get("ETag") != original.etag {
		t.Fatal("lost response did not retain exactly one update")
	}
	// A retained success is not permission to disclose it to a current reader.
	token = "reader"
	denied, err := c.UpdateNamespaceWithResponse(ctx, ns.JSON201.Id, params, body)
	must(t, err)
	if denied.StatusCode() != http.StatusForbidden || denied.ApplicationproblemJSONDefault == nil || denied.ApplicationproblemJSONDefault.Code != "FORBIDDEN" || denied.HTTPResponse.Header.Get("Idempotency-Replayed") != "" {
		t.Fatal("replay bypassed current write authorization")
	}
	token = "admin"
	changed := "Must not replace the committed update"
	conflict, err := c.UpdateNamespaceWithResponse(ctx, ns.JSON201.Id, params, sdk.UpdateNamespaceJSONRequestBody{DisplayName: &changed, Reason: body.Reason})
	must(t, err)
	if conflict.StatusCode() != http.StatusConflict || conflict.ApplicationproblemJSONDefault == nil || conflict.ApplicationproblemJSONDefault.Code != "IDEMPOTENCY_KEY_REUSED" {
		t.Fatal("changed command reused the retained binding")
	}
	replay, err := c.UpdateNamespaceWithResponse(ctx, ns.JSON201.Id, params, body)
	must(t, err)
	if replay.JSON200 == nil || !bytes.Equal(replay.Body, original.body) || replay.HTTPResponse.Header.Get("ETag") != original.etag || replay.HTTPResponse.Header.Get("Idempotency-Replayed") != "true" {
		t.Fatal("explicit original command did not recover the retained response")
	}
	after, err := c.GetNamespaceWithResponse(ctx, ns.JSON201.Id)
	must(t, err)
	if after.JSON200 == nil || !bytes.Equal(after.Body, readback.Body) || after.HTTPResponse.Header.Get("ETag") != original.etag || attempts.Load() != 4 {
		t.Fatal("refused or replayed requests changed state or added attempts", attempts.Load())
	}
}

		t.Fatal("add member", added.StatusCode())
	}
	replayed, err := admin.AddCollectionMemberWithResponse(ctx, created.JSON201.Id, params, body)
	must(t, err)
	if replayed.HTTPResponse.Header.Get("Idempotency-Replayed") != "true" || !bytes.Equal(replayed.Body, added.Body) {
		t.Fatal("member replay changed")
	}
	page, err := reader.ListCollectionMembersWithResponse(ctx, created.JSON201.Id, &sdk.ListCollectionMembersParams{Limit: ptr(1)})
	must(t, err)
	if page.JSON200 == nil || len(page.JSON200.Items) != 1 || page.JSON200.NextCursor != nil || page.JSON200.Items[0].Listing.Id != body.ListingId || page.JSON200.Items[0].Release.Id != record.JSON200.Release.Id {
		t.Fatal("member summary identities", page.StatusCode())
	}
	if page.HTTPResponse.Header.Get("Cache-Control") != "no-store" || page.HTTPResponse.Header.Get("ETag") != "" {
		t.Fatal("member cache policy")
	}
	filtered, err := reader.ListCollectionMembersWithResponse(ctx, created.JSON201.Id, &sdk.ListCollectionMembersParams{Text: ptr(record.JSON200.ListingRevision.DisplayName), Kinds: &[]sdk.ResourceKind{"agent", "skill"}})
	must(t, err)
	if filtered.JSON200 == nil || len(filtered.JSON200.Items) != 1 || filtered.JSON200.Items[0].Listing.Id != body.ListingId {
		t.Fatal("typed collection filters", filtered.StatusCode())
	}
	excluded, err := reader.ListCollectionMembersWithResponse(ctx, created.JSON201.Id, &sdk.ListCollectionMembersParams{Text: ptr("collection-filter-no-match"), Kinds: &[]sdk.ResourceKind{"agent", "skill"}})
	must(t, err)
	if excluded.JSON200 == nil || len(excluded.JSON200.Items) != 0 || excluded.JSON200.NextCursor != nil {
		t.Fatal("collection text filter ignored", excluded.StatusCode())
	}
	wrongKind, err := reader.ListCollectionMembersWithResponse(ctx, created.JSON201.Id, &sdk.ListCollectionMembersParams{Kinds: &[]sdk.ResourceKind{"mcp_server", "plugin"}})
	must(t, err)
	if wrongKind.JSON200 == nil || len(wrongKind.JSON200.Items) != 0 {
		t.Fatal("collection kind filter ignored", wrongKind.StatusCode())
	}
	directory, err := reader.ListCollectionsWithResponse(ctx, &sdk.ListCollectionsParams{NamespaceId: ptr(record.JSON200.Resource.NamespaceId), Limit: ptr(100)})
	must(t, err)
	if directory.JSON200 == nil || len(directory.JSON200.Items) != 1 || directory.JSON200.Items[0].Id != created.JSON201.Id || directory.JSON200.NextCursor != nil {
		t.Fatal("collection directory", directory.StatusCode())
	}
	metadata, err := reader.GetCollectionWithResponse(ctx, created.JSON201.Id)
	must(t, err)
	if metadata.JSON200 == nil || metadata.JSON200.Name != "Go SDK collection" || metadata.JSON200.Description != "Original note" || metadata.JSON200.Revision != 2 {
		t.Fatal("collection metadata", metadata.StatusCode())
	}
	updated, err := admin.UpdateCollectionWithResponse(ctx, created.JSON201.Id, &sdk.UpdateCollectionParams{IdempotencyKey: "go-update-collection", IfMatch: metadata.HTTPResponse.Header.Get("ETag")}, sdk.UpdateCollectionJSONRequestBody{Name: ptr("Updated Go collection"), Description: ptr("")})
	must(t, err)
	if updated.JSON200 == nil || updated.JSON200.Revision != 3 {
		t.Fatal("collection update", updated.StatusCode())
	}
	stale, err := admin.UpdateCollectionWithResponse(ctx, created.JSON201.Id, &sdk.UpdateCollectionParams{IdempotencyKey: "go-stale-collection", IfMatch: metadata.HTTPResponse.Header.Get("ETag")}, sdk.UpdateCollectionJSONRequestBody{Name: ptr("Stale")})
	must(t, err)
	if stale.StatusCode() != 412 {
		t.Fatal("stale collection update", stale.StatusCode())
	}
	denied, err := reader.DeleteCollectionWithResponse(ctx, created.JSON201.Id, &sdk.DeleteCollectionParams{IdempotencyKey: "go-reader-delete-collection", IfMatch: updated.HTTPResponse.Header.Get("ETag")})
	must(t, err)
	if denied.StatusCode() != 403 {
		t.Fatal("reader deleted collection", denied.StatusCode())
	}
	readback, err := reader.GetCollectionWithResponse(ctx, created.JSON201.Id)
	must(t, err)
	if readback.JSON200 == nil || readback.JSON200.Name != "Updated Go collection" || readback.JSON200.Description != "" || readback.JSON200.Revision != 3 {
		t.Fatal("collection update/refusal readback", readback.StatusCode())
	}
	removed, err := admin.RemoveCollectionMemberWithResponse(ctx, created.JSON201.Id, body.ListingId, &sdk.RemoveCollectionMemberParams{IdempotencyKey: "go-remove-member", IfMatch: updated.HTTPResponse.Header.Get("ETag")})
	must(t, err)
	if removed.JSON200 == nil || removed.JSON200.Revision != 4 {
		t.Fatal("remove member", removed.StatusCode())
	}
	empty, err := reader.ListCollectionMembersWithResponse(ctx, created.JSON201.Id, nil)
	must(t, err)
	if empty.JSON200 == nil || len(empty.JSON200.Items) != 0 || empty.JSON200.NextCursor != nil {
		t.Fatal("removed member remained visible", empty.StatusCode())
	}
	deleted, err := admin.DeleteCollectionWithResponse(ctx, created.JSON201.Id, &sdk.DeleteCollectionParams{IdempotencyKey: "go-delete-collection", IfMatch: removed.HTTPResponse.Header.Get("ETag")})
	must(t, err)
	if deleted.JSON200 == nil || deleted.JSON200.Revision != 5 {
		t.Fatal("delete collection", deleted.StatusCode())
	}
	absent, err := reader.GetCollectionWithResponse(ctx, created.JSON201.Id)
	must(t, err)
	if absent.StatusCode() != 404 {
		t.Fatal("deleted collection retained", absent.StatusCode())
	}
	preserved, err := reader.GetCatalogRecordWithResponse(ctx, releaseID)
	must(t, err)
	if preserved.JSON200 == nil || preserved.JSON200.Release.Id != record.JSON200.Release.Id {
		t.Fatal("curation changed source release", preserved.StatusCode())
	}

	paging := seedReviewedCollection(t, f, corpus)
	id := decode[sdk.UUID](t, []byte(strconv.Quote(paging.ID)))
	for _, selection := range paging.Selections {
		query := sdk.ListCollectionMembersParams{Limit: ptr(1), Text: &selection.Text, Kinds: &[]sdk.ResourceKind{"agent", "skill"}}
		for index, want := range selection.ListingIDs {
			page, err := reader.ListCollectionMembersWithResponse(ctx, id, &query)
			must(t, err)
			if page.JSON200 == nil || len(page.JSON200.Items) != 1 || page.JSON200.Items[0].Listing.Id.String() != want {
				t.Fatalf("filtered members page %d: expected Listing %s, HTTP %d", index+1, want, page.StatusCode())
			}
			query.Cursor = page.JSON200.NextCursor
			if index+1 < len(selection.ListingIDs) && (query.Cursor == nil || *query.Cursor == "") {
				t.Fatal("first filtered member page has no continuation")
			}
		}
		if query.Cursor != nil {
			t.Fatal("last filtered member page has a continuation")
		}
	}
}

// The installed Go SDK traverses actual maintenance pages and independent denials.
func TestManagedSDKValidationMaintenance(t *testing.T) {
	_, corpus := newReviewedCorpus(t)
	admin, err := sdk.New(sdk.Config{Origin: corpus.Origin, AllowLoopbackHTTP: true, Token: "admin"})
	must(t, err)
	defer admin.Close()
	reader, err := sdk.New(sdk.Config{Origin: corpus.Origin, AllowLoopbackHTTP: true, Token: "reader"})
	must(t, err)
	defer reader.Close()
	record, err := admin.GetCatalogRecordWithResponse(t.Context(), decode[sdk.CatalogRecordId](t, []byte(strconv.Quote(corpus.Cases[0].ReleaseID))))
	must(t, err)
	if record.JSON200 == nil {
		t.Fatal(record.StatusCode())
	}
	body, err := json.Marshal(map[string]any{"namespaceId": record.JSON200.Resource.NamespaceId, "kind": "agent", "canonicalIdentifier": "urn:air:catalog.example.com:go:maintenance"})
	must(t, err)
	resource, err := admin.CreateResourceWithResponse(t.Context(), &sdk.CreateResourceParams{IdempotencyKey: "go-maintenance-resource"}, decode[sdk.CreateResourceJSONRequestBody](t, body))
	must(t, err)
	if resource.JSON201 == nil {
		t.Fatal(resource.StatusCode())
	}
	want := map[string]bool{}
	for i := 0; i < 2; i++ {
		body, err := json.Marshal(map[string]any{"version": "79.0." + strconv.Itoa(i), "representations": []map[string]any{{"role": "primary", "mediaType": "application/a2a-agent-card+json", "data": map[string]any{"name": "Incomplete maintenance fixture"}}}})
		must(t, err)
		version, err := admin.CreateResourceVersionWithResponse(t.Context(), resource.JSON201.Id, &sdk.CreateResourceVersionParams{IdempotencyKey: "go-maintenance-version-" + strconv.Itoa(i)}, decode[sdk.CreateResourceVersionJSONRequestBody](t, body))
		must(t, err)
		if version.JSON201 == nil || version.JSON201.Evidence.State != "blocked" {
			t.Fatal("blocked control", version.StatusCode())
		}
		want[version.JSON201.Revision.Id.String()] = true
	}
	state := sdk.ListValidationMaintenanceParamsState("blocked")
	params := &sdk.ListValidationMaintenanceParams{State: &state, HorizonHours: ptr(24), Limit: ptr(1)}
	first, err := admin.ListValidationMaintenanceWithResponse(t.Context(), params)
	must(t, err)
	if first.JSON200 == nil || len(first.JSON200.Items) != 1 || first.JSON200.NextCursor == nil || !want[first.JSON200.Items[0].RevisionId.String()] || first.HTTPResponse.Header.Get("Cache-Control") != "no-store" {
		t.Fatal("first maintenance page", first.StatusCode())
	}
	params.Cursor = first.JSON200.NextCursor
	second, err := admin.ListValidationMaintenanceWithResponse(t.Context(), params)
	must(t, err)
	if second.JSON200 == nil || len(second.JSON200.Items) != 1 || second.JSON200.NextCursor != nil || !want[second.JSON200.Items[0].RevisionId.String()] || first.JSON200.Items[0].RevisionId == second.JSON200.Items[0].RevisionId || !first.JSON200.ObservedAt.Equal(second.JSON200.ObservedAt) {
		t.Fatal("maintenance continuation", second.StatusCode())
	}
	params.Cursor = nil
	denied, err := reader.ListValidationMaintenanceWithResponse(t.Context(), params)
	must(t, err)
	if denied.JSON200 == nil || len(denied.JSON200.Items) != 0 {
		t.Fatal("reader disclosed validation maintenance", denied.StatusCode())
	}
	params.HorizonHours = ptr(169)
	invalid, err := admin.ListValidationMaintenanceWithResponse(t.Context(), params)
	must(t, err)
	if invalid.StatusCode() != 400 || invalid.ApplicationproblemJSONDefault == nil || invalid.ApplicationproblemJSONDefault.Code != "INVALID_ARGUMENT" {
		t.Fatal("invalid horizon", invalid.StatusCode())
	}
}
